          Estructura mínima

          // State: el episodio delega su comportamiento en el estado actual
interface EstadoEpisodio {
    boolean puedeEgresar();
}
class EnObservacion implements EstadoEpisodio {
    public boolean puedeEgresar() { return true; }
}
class AtencionPrioritaria implements EstadoEpisodio {
    public boolean puedeEgresar() { return false; }
}

// Observer: quien se suscribe recibe cada evento
interface Observador { void actualizar(Evento ev); }

class Episodio {
    private EstadoEpisodio estado = new Admitido();
    private final List<Observador> suscriptores = new ArrayList<>();

    void suscribir(Observador o) { suscriptores.add(o); }

    void cambiarEstado(EstadoEpisodio nuevo) {
        estado = nuevo;
        notificar(new Evento("Estado: " + nuevo));
    }
    void notificar(Evento ev) {
        for (Observador o : suscriptores) o.actualizar(ev);
    }
}

// Strategy: el manejo cambia sin tocar el resto del sistema
interface EstrategiaManejo { List<Orden> planificar(Paciente p); }
class ManejoConExamenes implements EstrategiaManejo {
    public List<Orden> planificar(Paciente p) {
        return List.of(new OrdenRx("columna lumbar"), new OrdenMedicamento("analgesia"));
    }
}

// Command: cada orden es un objeto que se ejecuta, registra o deshace
interface Orden { void ejecutar(); void registrar(); void deshacer(); }

          Nota didáctica: los patrones no ocupan casillas fijas

          En los cómics, cada sección muestra un patrón asociado a una etapa. Esa presentación es estática por efectos didácticos. En el sistema real, los patrones se activan donde hacen falta: State acompaña a Diego desde que entra hasta que sale, y Observer se dispara cada vez que algo cambia, no solo en la observación. Así sucesivamente con Facade, Strategy y Command.
